rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isMember(memberId) {
      return signedIn() && exists(/databases/$(database)/documents/families/sartors/members/$(memberId)/devices/$(request.auth.uid));
    }
    function isKnownFamilyDevice() {
      return isMember('tamara') || isMember('matt') || isMember('tommy') || isMember('charlie');
    }
    function isEditor() { return isMember('tamara') || isMember('matt') || isMember('tommy'); }

    match /families/sartors/members/{memberId} {
      allow read: if signedIn();
      allow create, update: if signedIn() && request.auth.uid == request.resource.data.uid;
      match /devices/{deviceId} {
        allow read: if signedIn();
        allow create, update: if signedIn() && deviceId == request.auth.uid && request.resource.data.uid == request.auth.uid;
        allow delete: if signedIn() && deviceId == request.auth.uid;
      }
    }
    match /families/sartors/events/{id} { allow read: if isKnownFamilyDevice(); allow create, update, delete: if isEditor(); }
    match /families/sartors/dinners/{id} { allow read: if isKnownFamilyDevice(); allow create, update, delete: if isEditor(); }
    match /families/sartors/lists/{id} { allow read: if isKnownFamilyDevice(); allow create, update, delete: if isEditor(); }
    match /families/sartors/requests/{id} {
      allow read: if isKnownFamilyDevice();
      allow create: if isKnownFamilyDevice() && request.resource.data.member in ['tamara','matt','tommy','charlie'];
      allow update, delete: if isEditor();
    }
  }
}
